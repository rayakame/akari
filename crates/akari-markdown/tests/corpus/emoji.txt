# Unicode emoji, :shortcodes: and jumbo messages (R§10, R§12, D66). Emoji show as text; a
# message of 1 to 27 emoji (Unicode or custom) and whitespace is jumbo.

== one emoji is jumbo
😀
--
p: "😀"
jumbo

== emoji with spaces are jumbo
😀 😀
--
p: "😀 😀"
jumbo

== emoji next to each other are jumbo
😀😀
--
p: "😀😀"
jumbo

== emoji on two lines are jumbo
input: "😀\n😀"
--
p: "😀" br "😀"
jumbo

== spaces around emoji
input: " 😀 "
--
p: " 😀 "
jumbo

== an escaped emoji isn't one
\😀
--

== a combining mark after an emoji is text
input: "😀\u{301}"
--
p: "😀́"

== a zero-width space after an emoji is text
input: "😀\u{200b}"
--
p: "😀\u{200b}"

== italics win over an asterisk keycap
*️⃣ a*
--
p: {i}"️⃣ a"

== a keycap before a period isn't a list
1️⃣. a
--
p: "1️⃣. a"

== a hash keycap isn't a heading
#️⃣ a
--
p: "#️⃣ a"

== observe: a styled emoji isn't jumbo
**😀**
--
p: {b}"😀"

== observe: a mention cancels jumbo
😀 <@1>
--
p: "😀 " @user:1

== observe: a timestamp cancels jumbo
😀 <t:0>
--
p: "😀 " time:0:f

== observe: a spoiler cancels jumbo
||😀||
--
p{sp0}: {sp0}"😀"

== shortcodes stay text
:smile:
--
p: ":smile:"

== a shortcode's underscores don't start italics
_:a_b:_
--
p: "_:a_b:_"

== a shortcode is read as a whole
x :a_b: y_
--
p: "x :a_b: y_"
